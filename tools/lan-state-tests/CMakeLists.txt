cmake_minimum_required(VERSION 3.14)
project(voidframe_lan_state_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core)
add_executable(voidframe_lan_state_tests main.cpp ../../CppProject/Lan/LanState.cpp)
target_include_directories(voidframe_lan_state_tests PRIVATE ../../CppProject/Lan)
target_link_libraries(voidframe_lan_state_tests PRIVATE Qt5::Core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(voidframe_lan_state_tests PRIVATE -Wall -Wextra -Werror=return-type)
endif()
enable_testing()
add_test(NAME lan_authority COMMAND voidframe_lan_state_tests)
set_tests_properties(lan_authority PROPERTIES TIMEOUT 20)
