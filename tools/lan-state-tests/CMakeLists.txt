cmake_minimum_required(VERSION 3.14)
project(LanChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Network Gui)
set(LAN ${CMAKE_CURRENT_SOURCE_DIR}/../../CppProject/Lan)
add_library(lancore STATIC
  ${LAN}/LanState.cpp ${LAN}/LanTransactions.cpp
  ${LAN}/LanSession.cpp ${LAN}/LanReplica.cpp ${LAN}/LanPresence.cpp)
target_include_directories(lancore PUBLIC ${LAN})
target_link_libraries(lancore PUBLIC Qt5::Core Qt5::Network Qt5::Gui)
configure_file(main.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/main.cpp COPYONLY)
configure_file(network.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/network.cpp COPYONLY)
add_executable(lanchecks ${CMAKE_CURRENT_BINARY_DIR}/main.cpp)
add_executable(lannetwork ${CMAKE_CURRENT_BINARY_DIR}/network.cpp)
target_link_libraries(lanchecks PRIVATE lancore)
target_link_libraries(lannetwork PRIVATE lancore)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lancore PRIVATE -Wall -Wextra -Werror=return-type)
endif()
enable_testing()
add_test(NAME lan_authority COMMAND lanchecks)
add_test(NAME lan_network COMMAND lannetwork)
set_tests_properties(lan_authority lan_network PROPERTIES TIMEOUT 30)
