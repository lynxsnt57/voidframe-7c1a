cmake_minimum_required(VERSION 3.14)
project(LanChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core)
configure_file(main.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/main.cpp COPYONLY)
add_executable(lanchecks ${CMAKE_CURRENT_BINARY_DIR}/main.cpp
  ../../CppProject/Lan/LanState.cpp
  ../../CppProject/Lan/LanTransactions.cpp)
target_include_directories(lanchecks PRIVATE ../../CppProject/Lan)
target_link_libraries(lanchecks PRIVATE Qt5::Core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lanchecks PRIVATE -Wall -Wextra -Werror=return-type)
endif()
enable_testing()
add_test(NAME lan_authority COMMAND lanchecks)
set_tests_properties(lan_authority PROPERTIES TIMEOUT 20)
